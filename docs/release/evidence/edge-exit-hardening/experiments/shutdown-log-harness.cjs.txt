function sha256(contents) {
  return createHash('sha256').update(contents).digest('hex');
}

function readBaseline([relativePath, gitBlob]) {
  const filename = path.join(repo, relativePath);
  const source = fs.readFileSync(filename, 'utf8').replace(/\r\n/g, '\n');
  const digest = createHash('sha1').update('blob ' + Buffer.byteLength(source) + '\0').update(source).digest('hex');
  assert.equal(digest, gitBlob, 'Source changed since ' + baseline + ': ' + relativePath);
  return { filename, source, relativePath, gitBlob, sourceSha256: sha256(source) };
}

function replaceOnce(source, anchor, replacement) {
  const offset = source.indexOf(anchor);
  assert.ok(offset >= 0 && source.lastIndexOf(anchor) === offset, 'Expected exactly one source anchor: ' + anchor);
  return source.slice(0, offset) + replacement + source.slice(offset + anchor.length);
}

function createShutdownLogObservation() {
  const maxScannedChars = 1024 * 1024, maxLineChars = 4096, maxMarkers = 8;
  const markers = [];
  let scannedChars = 0, ignoredLongLines = 0, extraMarkers = 0;
  let partial = '', skippingLine = false, truncated = false;
  return {
    receive(chunk, rootPid, elapsedMs) {
      const remaining = maxScannedChars - scannedChars;
      if (chunk.length > remaining) truncated = true;
      const text = chunk.slice(0, remaining);
      scannedChars += text.length;
      for (let start = 0; start < text.length;) {
        const newline = text.indexOf('\n', start);
        const end = newline < 0 ? text.length : newline;
        if (!skippingLine) {
          if (partial.length + end - start > maxLineChars) {
            skippingLine = true;
            ignoredLongLines++;
            partial = '';
          } else {
            partial += text.slice(start, end);
          }
        }
        if (newline < 0) break;
        if (!skippingLine) {
          const fields = markerLine.exec(partial)?.groups;
          if (fields && Number(fields.pid) === rootPid) {
            if (markers.length < maxMarkers) {
              markers.push({
                marker: 'ExitIgnoreUnloadHandlers',
                elapsedMs,
                browserLocalWallClock: fields.clock,
                source: 'application_lifetime.cc',
                sourceLine: Number(fields.colonLine ?? fields.parenthesizedLine),
                prefixFormat: fields.colonLine === undefined ? 'filename(line)' : 'filename:line',
              });
            } else extraMarkers++;
          }
        }
        partial = '';
        skippingLine = false;
        start = newline + 1;
      }
    },
    snapshot() {
      return {
        status: markers.length ? 'Observed' : 'Unverified',
        timebase: 'node-monotonic-ms at complete stderr line receipt',
        markers, scannedChars, ignoredLongLines, extraMarkers, truncated,
        incompleteLine: partial.length > 0 || skippingLine,
        maxScannedChars, maxLineChars, maxMarkers,
      };
    },
  };
}

function urlCategory(url) {
  if (url === 'about:blank') return 'about:blank';
  if (typeof url !== 'string' || !url) return 'missing';
  if (url.startsWith('about:')) return 'about-other';
  if (/^(?:edge|chrome|chrome-untrusted|devtools):/i.test(url)) return 'browser-internal';
  if (/^file:/i.test(url)) return 'file';
  if (/^https?:/i.test(url)) return 'http-or-https';
  return 'other';
}

function checkShutdownLogParser() {
  const rootPid = 15628;
  const clock = '1119/013002.894';
  const prefix = '[' + rootPid + ':10780:' + clock + ':VERBOSE1:';
  const marker = 'ExitIgnoreUnloadHandlers';
  const colon = prefix + 'chrome\\browser\\lifetime\\application_lifetime.cc:81] ' + marker;
  const parenthesized = prefix + 'application_lifetime.cc(81)] ' + marker;
  const makeExpected = (elapsedMs, prefixFormat) => ({
    marker, elapsedMs, browserLocalWallClock: clock,
    source: 'application_lifetime.cc', sourceLine: 81, prefixFormat,
  });
  const accepted = createShutdownLogObservation();
  accepted.receive(colon.slice(0, 36), rootPid, 10);
  accepted.receive(colon.slice(36) + '\r', rootPid, 11);
  assert.equal(accepted.snapshot().status, 'Unverified');
  accepted.receive('\n' + parenthesized + '\n', rootPid, 12);
  assert.deepEqual(accepted.snapshot().markers, [
    makeExpected(12, 'filename:line'), makeExpected(12, 'filename(line)'),
  ]);
  assert.equal(accepted.snapshot().status, 'Observed');

  const denied = createShutdownLogObservation();
  const invalid = [
    colon.replace('[' + rootPid + ':', '[15629:'),
    colon.replace('application_lifetime.cc', 'other_application_lifetime.cc'),
    colon.replace('application_lifetime.cc', 'fallback_task_provider.cc'),
    colon.replace(':VERBOSE1:', ':ERROR:'),
    colon.replace(':81]', '(81]'),
    colon.replace(marker, 'Unexpected text ' + marker),
    colon + ' trailing text',
  ];
  denied.receive(invalid.join('\n') + '\n', rootPid, 13);
  assert.deepEqual(denied.snapshot().markers, []);
  assert.equal(denied.snapshot().status, 'Unverified');

  const lineBoundary = createShutdownLogObservation();
  const { maxLineChars, maxScannedChars, maxMarkers } = lineBoundary.snapshot();
  lineBoundary.receive(colon.padEnd(maxLineChars, ' '), rootPid, 14);
  assert.equal(lineBoundary.snapshot().incompleteLine, true);
  lineBoundary.receive('\n' + colon.padEnd(maxLineChars + 1, ' ') + '\n' + parenthesized + '\n', rootPid, 15);
  assert.deepEqual(lineBoundary.snapshot().markers, [
    makeExpected(15, 'filename:line'), makeExpected(15, 'filename(line)'),
  ]);
  assert.equal(lineBoundary.snapshot().ignoredLongLines, 1);

  const atScanLimit = createShutdownLogObservation();
  const filler = 'x'.repeat(maxScannedChars - colon.length - 2) + '\n';
  atScanLimit.receive(filler + colon + '\n', rootPid, 16);
  assert.deepEqual(atScanLimit.snapshot().markers, [makeExpected(16, 'filename:line')]);
  assert.equal(atScanLimit.snapshot().truncated, false);
  const truncated = createShutdownLogObservation();
  truncated.receive('x' + filler + colon + '\n', rootPid, 17);
  assert.equal(truncated.snapshot().status, 'Unverified');
  assert.equal(truncated.snapshot().truncated, true);
  assert.equal(truncated.snapshot().incompleteLine, true);

  const countBoundary = createShutdownLogObservation();
  countBoundary.receive((colon + '\n').repeat(maxMarkers + 1), rootPid, 18);
  assert.equal(countBoundary.snapshot().markers.length, maxMarkers);
  assert.equal(countBoundary.snapshot().extraMarkers, 1);
  return ['both-prefixes', 'exact-allowlist', 'line-boundary', 'scan-boundary', 'marker-count-boundary'];
}

function startupTargetSnapshot(events) {
  const targets = new Map();
  for (const event of events) {
    if (event.method === 'Target.targetDestroyed') {
      targets.delete(event.params?.targetId);
    } else if (event.method === 'Target.targetCreated' || event.method === 'Target.targetInfoChanged') {
      const target = event.params?.targetInfo;
      if (typeof target?.targetId === 'string') targets.set(target.targetId, target);
    }
  }
  const pages = [...targets.values()].filter(target => target.type === 'page');
  const pageUrlCategories = {};
  for (const page of pages) {
    const category = urlCategory(page.url);
    pageUrlCategories[category] = (pageUrlCategories[category] || 0) + 1;
  }
  return {
    source: 'existing discovery-event cache at discovery acknowledgement, before owned target creation',
    observedTargets: targets.size, observedPages: pages.length, pageUrlCategories,
    potentiallyTruncatedEventHistory: events.length >= 200,
  };
}

function instrumentRunner(source) {
  let instrumented = replaceOnce(source, "const deferred = () =>", [
    "const { createShutdownLogObservation, startupTargetSnapshot } = require('" + diagnosticsDependency + "');",
    'const deferred = () =>',
  ].join('\n'));
  instrumented = replaceOnce(instrumented, "  let stdout = '', stderr = '', child,",
    "  const shutdownLog = createShutdownLogObservation();\n  let stdout = '', stderr = '', child,");
  instrumented = replaceOnce(instrumented, "child.stderr.on('data', value => { stderr += value;",
    "child.stderr.on('data', value => { shutdownLog.receive(value, child.pid, elapsed()); stderr += value;");
  instrumented = replaceOnce(instrumented, '    metadata.stdoutBytes = Buffer.byteLength(stdout);',
    '    metadata.shutdownLogObservation = shutdownLog.snapshot();\n    metadata.stdoutBytes = Buffer.byteLength(stdout);');
  instrumented = replaceOnce(instrumented,
    "    if (problem) evidenceWrites.push(fsp.writeFile(stem + '.stdout.log', stdout), fsp.writeFile(stem + '.stderr.log', stderr));",
    '    // Diagnostic run retains sanitized marker evidence and the original byte counts.');
  instrumented = replaceOnce(instrumented, "'--window-size=1280,720', 'about:blank'],",
    "'--window-size=1280,720', " + flags.map(flag => JSON.stringify(flag)).join(', ') + ", 'about:blank'],");
  const discovery = "      await connection.send('Target.setDiscoverTargets', { discover: true });";
  return replaceOnce(instrumented, discovery,
    discovery + "\n      context.mark('shutdown-log-startup-targets-observed', startupTargetSnapshot(connection.events));");
}

function compile(filename, source, dependencies) {
  const compiled = new Module(filename, module);
  compiled.filename = filename;
  compiled.paths = Module._nodeModulePaths(path.dirname(filename));
  const originalRequire = compiled.require.bind(compiled);
  compiled.require = name => Object.hasOwn(dependencies, name) ? dependencies[name] : originalRequire(name);
  compiled._compile(source, filename);
  return compiled.exports;
}
